    <script>
        function checkAnswer(isCorrect, element) {
            // Сброс стилей кнопок
            document.querySelectorAll('.quiz-btn').forEach(btn => {
                btn.classList.remove('border-red-500', 'border-green-500', 'bg-red-500/10', 'bg-green-500/10');
            });

            const feedbackPanel = document.getElementById('feedback-panel');
            const feedbackTitle = document.getElementById('feedback-title');
            const feedbackText = document.getElementById('feedback-text');

            feedbackPanel.classList.remove('hidden', 'bg-red-500/10', 'border-red-500', 'text-red-400', 'bg-green-500/10', 'border-green-500', 'text-green-400');

            if (isCorrect) {
                element.classList.add('border-green-500', 'bg-green-500/10');
                feedbackPanel.classList.add('bg-green-500/10', 'border-green-500', 'text-green-400');
                feedbackTitle.innerText = "Correct! (C1 Vocabulary)";
                feedbackText.innerText = "The use of 'intended' shifts the blame away from the speaker in a formal context, masking guilt.";
            } else {
                element.classList.add('border-red-500', 'bg-red-500/10');
                feedbackPanel.classList.add('bg-red-500/10', 'border-red-500', 'text-red-400');
                feedbackTitle.innerText = "Incorrect register.";
                feedbackText.innerText = "This option is grammatically correct but too informal (B1/B2) for this specific high-stakes corporate negotiation.";
            }
        }
    </script>
